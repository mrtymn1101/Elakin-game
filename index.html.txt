    <script>
        // Harfler ve Renk Haritası
        const letters = [
            { char: 'e', class: 'e-color' },
            { char: 'l', class: 'l-color' },
            { char: 'a', class: 'a-color' },
            { char: 'k', class: 'k-color' },
            { char: 'i', class: 'i-color' },
            { char: 'n', class: 'n-color' }
        ];

        // ELAKİN Grubundaki TÜM Olası 2'li Heceler (36 Adet)
        const allSyllables = [];
        letters.forEach(l1 => {
            letters.forEach(l2 => {
                allSyllables.push(l1.char + l2.char);
            });
        });

        let currentCarSyllable = "el";
        const synth = window.speechSynthesis;

        // Web Audio API ile Sentetik Araba Motor Sesi Oluşturucu
        let audioCtx = null;
        let oscillator = null;

        function startCarSound() {
            try {
                if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
                if (audioCtx.state === 'suspended') audioCtx.resume();
                
                oscillator = audioCtx.createOscillator();
                const gainNode = audioCtx.createGain();
                
                oscillator.type = 'sawtooth'; // Motor benzeri tırtıklı ses
                oscillator.frequency.setValueAtTime(60, audioCtx.currentTime); // Düşük frekans
                
                gainNode.gain.setValueAtTime(0.1, audioCtx.currentTime);
                
                oscillator.connect(gainNode);
                gainNode.connect(audioCtx.destination);
                oscillator.start();
            } catch(e) {}
        }

        function stopCarSound() {
            if (oscillator) {
                try {
                    oscillator.stop();
                    oscillator.disconnect();
                } catch(e) {}
                oscillator = null;
            }
        }

        // --- SAYFA BAŞLATMA ---
        window.onload = () => {
            renderSyllableChips();
            setupCarDrive();
            renderFreeMode();
            selectCarSyllable('el');
        };

        function switchTab(tabId) {
            document.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('active'));
            document.querySelectorAll('.tab-content').forEach(content => content.classList.remove('active'));
            
            event.target.classList.add('active');
            document.getElementById(tabId).classList.add('active');
        }

        // --- ARABA MODU MANTIĞI ---
        function renderSyllableChips() {
            const container = document.getElementById('syllableChips');
            container.innerHTML = '';
            allSyllables.forEach(syl => {
                const chip = document.createElement('div');
                chip.className = `syllable-chip ${syl === currentCarSyllable ? 'active' : ''}`;
                chip.innerText = syl.toUpperCase();
                chip.onclick = () => selectCarSyllable(syl);
                container.appendChild(chip);
            });
        }

        function selectCarSyllable(syl) {
            currentCarSyllable = syl;
            const c1 = syl[0];
            const c2 = syl[1];

            const item1 = letters.find(l => l.char === c1);
            const item2 = letters.find(l => l.char === c2);

            const startEl = document.getElementById('carStartLetter');
            const endEl = document.getElementById('carEndLetter');

            startEl.innerText = c1.toUpperCase();
            startEl.className = `road-letter ${item1.class}`;

            endEl.innerText = c2.toUpperCase();
            endEl.className = `road-letter ${item2.class}`;

            document.getElementById('carResultText').innerHTML = `
                <span class="${item1.class}">${c1.toUpperCase()}</span><span class="${item2.class}">${c2.toUpperCase()}</span>
            `;

            renderSyllableChips();
            resetCarPosition();
        }

        function resetCarPosition() {
            const car = document.getElementById('draggableCar');
            car.style.left = '80px';
        }

        function setupCarDrive() {
            const car = document.getElementById('draggableCar');
            const road = document.getElementById('roadContainer');
            let isDragging = false;

            const onStart = (e) => {
                isDragging = true;
                startCarSound();
            };

            const onMove = (e) => {
                if (!isDragging) return;
                
                const clientX = e.touches ? e.touches[0].clientX : e.clientX;
                const roadRect = road.getBoundingClientRect();
                
                let newX = clientX - roadRect.left - 25; // Araba merkezleme
                
                const minX = 80; // Başlangıç harfi sınırı
                const maxX = roadRect.width - 120; // Bitiş harfi sınırı

                if (newX < minX) newX = minX;
                if (newX > maxX) newX = maxX;

                car.style.left = `${newX}px`;

                // Bitişe ulaştı mı?
                if (newX >= maxX - 5) {
                    isDragging = false;
                    stopCarSound();
                    speakCurrentCarSyllable();
                    resetCarPosition();
                }
            };

            const onEnd = () => {
                if (isDragging) {
                    isDragging = false;
                    stopCarSound();
                    resetCarPosition();
                }
            };

            // Fare Olayları
            car.addEventListener('mousedown', onStart);
            window.addEventListener('mousemove', onMove);
            window.addEventListener('mouseup', onEnd);

            // Dokunmatik (Mobil/Tablet) Olayları
            car.addEventListener('touchstart', onStart);
            window.addEventListener('touchmove', onMove);
            window.addEventListener('touchend', onEnd);
        }

        function speakCurrentCarSyllable() {
            speak(currentCarSyllable);
        }

        // --- SERBEST BİRLEŞTİRME MODU MANTIĞI ---
        let selectedFirst = null;
        let selectedSecond = null;

        function renderFreeMode() {
            renderButtons('firstLetters', 1);
            renderButtons('secondLetters', 2);
        }

        function renderButtons(containerId, slot) {
            const container = document.getElementById(containerId);
            container.innerHTML = '';
            letters.forEach(item => {
                const btn = document.createElement('button');
                btn.className = `letter-btn ${item.class}`;
                btn.innerText = item.char;
                btn.onclick = () => {
                    speak(item.char);
                    if (slot === 1) {
                        selectedFirst = item;
                        document.getElementById('slot1').innerText = item.char.toUpperCase();
                        document.getElementById('slot1').className = `slot active ${item.class}`;
                    } else {
                        selectedSecond = item;
                        document.getElementById('slot2').innerText = item.char.toUpperCase();
                        document.getElementById('slot2').className = `slot active ${item.class}`;
                    }
                    document.getElementById('combineBtn').disabled = !(selectedFirst && selectedSecond);
                };
                container.appendChild(btn);
            });
        }

        function combineLetters() {
            if (!selectedFirst || !selectedSecond) return;
            const syl = selectedFirst.char + selectedSecond.char;
            document.getElementById('resultText').innerHTML = `
                <span class="${selectedFirst.class}">${selectedFirst.char.toUpperCase()}</span><span class="${selectedSecond.class}">${selectedSecond.char.toUpperCase()}</span>
            `;
            speak(syl);
        }

        function speakResult() {
            if (selectedFirst && selectedSecond) speak(selectedFirst.char + selectedSecond.char);
        }

        function resetGame() {
            selectedFirst = null;
            selectedSecond = null;
            document.getElementById('slot1').innerText = '?';
            document.getElementById('slot1').className = 'slot';
            document.getElementById('slot2').innerText = '?';
            document.getElementById('slot2').className = 'slot';
            document.getElementById('resultText').innerText = '✨';
            document.getElementById('combineBtn').disabled = true;
        }

        // SESLENDİRME FONKSİYONU
        function speak(text) {
            if ('speechSynthesis' in window) {
                synth.cancel();
                const utterance = new SpeechSynthesisUtterance(text);
                utterance.lang = 'tr-TR';
                utterance.rate = 0.8;
                utterance.pitch = 1.1;
                synth.speak(utterance);
            }
        }
    </script>
